
-- TABLE: menu_cycles
CREATE TABLE `menu_cycles` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `pae_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `description` text DEFAULT NULL,
  `start_date` date NOT NULL,
  `end_date` date NOT NULL,
  `total_days` int(11) DEFAULT 20,
  `is_validated` tinyint(1) DEFAULT 0,
  `validated_by` int(11) DEFAULT NULL,
  `validated_at` timestamp NULL DEFAULT NULL,
  `status` enum('BORRADOR','ACTIVO','FINALIZADO') DEFAULT 'BORRADOR',
  `created_at` timestamp NOT NULL DEFAULT current_timestamp(),
  `updated_at` timestamp NOT NULL DEFAULT current_timestamp() ON UPDATE current_timestamp(),
  PRIMARY KEY (`id`),
  KEY `validated_by` (`validated_by`),
  KEY `idx_cycles_pae` (`pae_id`),
  CONSTRAINT `menu_cycles_ibfk_1` FOREIGN KEY (`pae_id`) REFERENCES `pae_programs` (`id`) ON DELETE CASCADE,
  CONSTRAINT `menu_cycles_ibfk_2` FOREIGN KEY (`validated_by`) REFERENCES `users` (`id`) ON DELETE SET NULL
) ENGINE=InnoDB AUTO_INCREMENT=117 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- TABLE: menus
CREATE TABLE `menus` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `pae_id` int(11) NOT NULL,
  `cycle_id` int(11) DEFAULT NULL,
  `name` varchar(255) NOT NULL,
  `day_number` int(11) DEFAULT NULL,
  `meal_type` enum('DESAYUNO','COMPLEMENTO ALIMENTARIO JORNADA DE LA TARDE','ALMUERZO','REFRIGERIO','REFRIGERIO REFORZADO INDUSTRIALIZADO','DESAYUNO INDUSTRIALIZADO PARA EMERGENCIAS') DEFAULT 'ALMUERZO',
  `ration_type_id` int(11) DEFAULT NULL,
  `age_group` enum('PREESCOLAR','PRIMARIA_A','PRIMARIA_B','SECUNDARIA','TODOS') DEFAULT 'TODOS',
  `has_dairy` tinyint(1) DEFAULT 0,
  `has_protein` tinyint(1) DEFAULT 0,
  `has_cereal` tinyint(1) DEFAULT 0,
  `has_fruit` tinyint(1) DEFAULT 0,
  `has_vegetable` tinyint(1) DEFAULT 0,
  `total_calories` decimal(10,2) DEFAULT 0.00,
  `total_proteins` decimal(10,2) DEFAULT 0.00,
  `total_carbohydrates` decimal(10,2) DEFAULT 0.00,
  `total_fats` decimal(10,2) DEFAULT 0.00,
  `total_iron` decimal(10,2) DEFAULT 0.00,
  `total_calcium` decimal(10,2) DEFAULT 0.00,
  `total_cost` decimal(10,2) DEFAULT 0.00,
  `preparation_instructions` text DEFAULT NULL,
  `allergen_warnings` text DEFAULT NULL,
  `status` enum('ACTIVO','INACTIVO') DEFAULT 'ACTIVO',
  `created_at` timestamp NOT NULL DEFAULT current_timestamp(),
  `updated_at` timestamp NOT NULL DEFAULT current_timestamp() ON UPDATE current_timestamp(),
  PRIMARY KEY (`id`),
  KEY `idx_menus_pae` (`pae_id`),
  KEY `idx_menus_cycle` (`cycle_id`),
  KEY `idx_menus_age` (`age_group`),
  CONSTRAINT `menus_ibfk_1` FOREIGN KEY (`pae_id`) REFERENCES `pae_programs` (`id`) ON DELETE CASCADE,
  CONSTRAINT `menus_ibfk_2` FOREIGN KEY (`cycle_id`) REFERENCES `menu_cycles` (`id`) ON DELETE SET NULL
) ENGINE=InnoDB AUTO_INCREMENT=476 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- TABLE: cycle_projections
CREATE TABLE `cycle_projections` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `cycle_id` int(11) NOT NULL,
  `branch_id` int(11) NOT NULL,
  `item_id` int(11) NOT NULL,
  `total_quantity` decimal(12,4) NOT NULL,
  `beneficiary_count` int(11) NOT NULL,
  `created_at` timestamp NOT NULL DEFAULT current_timestamp(),
  PRIMARY KEY (`id`),
  UNIQUE KEY `unique_projection` (`cycle_id`,`branch_id`,`item_id`),
  KEY `idx_proj_cycle` (`cycle_id`),
  KEY `idx_proj_branch` (`branch_id`),
  KEY `idx_proj_item` (`item_id`),
  CONSTRAINT `cycle_projections_ibfk_1` FOREIGN KEY (`cycle_id`) REFERENCES `menu_cycles` (`id`) ON DELETE CASCADE,
  CONSTRAINT `cycle_projections_ibfk_2` FOREIGN KEY (`branch_id`) REFERENCES `school_branches` (`id`) ON DELETE CASCADE,
  CONSTRAINT `cycle_projections_ibfk_3` FOREIGN KEY (`item_id`) REFERENCES `items` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB AUTO_INCREMENT=326 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
