=== SCHEMA: cycle_template_days ===
CREATE TABLE `cycle_template_days` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `template_id` int(11) NOT NULL,
  `day_number` int(11) NOT NULL,
  `meal_type` enum('DESAYUNO','COMPLEMENTO ALIMENTARIO JORNADA DE LA TARDE','ALMUERZO','REFRIGERIO','REFRIGERIO REFORZADO INDUSTRIALIZADO','DESAYUNO INDUSTRIALIZADO PARA EMERGENCIAS') NOT NULL,
  `ration_type_id` int(11) DEFAULT NULL,
  `recipe_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `unique_meal_per_day` (`template_id`,`day_number`,`meal_type`),
  KEY `recipe_id` (`recipe_id`),
  CONSTRAINT `cycle_template_days_ibfk_1` FOREIGN KEY (`template_id`) REFERENCES `cycle_templates` (`id`) ON DELETE CASCADE,
  CONSTRAINT `cycle_template_days_ibfk_2` FOREIGN KEY (`recipe_id`) REFERENCES `recipes` (`id`)
) ENGINE=InnoDB AUTO_INCREMENT=87 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci
